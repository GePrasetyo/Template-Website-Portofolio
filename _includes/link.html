{%- comment -%} include.l: { kind: link | email | resume, label, url }. include.arrow adds the ↗ icon. Renders nothing when the link has no target (no email or résumé set in the profile, empty URL). {%- endcomment -%} {%- assign ext = true -%} {%- case include.l.kind -%} {%- when 'email' -%} {%- if site.data.profile.email -%}{%- assign href = 'mailto:' | append: site.data.profile.email -%}{%- else -%}{%- assign href = '' -%}{%- endif -%} {%- assign label = include.l.label | default: site.data.profile.email -%} {%- assign ext = false -%} {%- when 'resume' -%} {%- if site.data.profile.resume -%}{%- assign href = site.data.profile.resume | relative_url -%}{%- else -%}{%- assign href = '' -%}{%- endif -%} {%- assign label = include.l.label | default: 'Résumé (PDF)' -%} {%- else -%} {%- assign href = include.l.url | strip -%} {%- assign label = include.l.label | default: href -%} {%- assign head = href | slice: 0 -%} {%- if head == '/' -%}{%- assign href = href | relative_url -%}{%- endif -%} {%- if href contains 'mailto:' -%}{%- assign ext = false -%}{%- endif -%} {%- endcase -%} {%- if href != '' %} {{ label | escape }}{% if include.arrow %} {% endif %} {%- endif -%}
